  <script>

    const foodInputs = document.querySelectorAll(
      'input[name="food"]'
    );

    const dateInput = document.getElementById("date");
    const timeInput = document.getElementById("time");

    const confirmButton = document.getElementById("confirm");
    const hint = document.getElementById("hint");

    const mainPage = document.getElementById("mainPage");
    const successPage = document.getElementById("successPage");

    const finalFood = document.getElementById("finalFood");
    const finalDate = document.getElementById("finalDate");
    const finalTime = document.getElementById("finalTime");


    // Impedisce di scegliere una data nel passato
    const today = new Date();

    const year = today.getFullYear();
    const month = String(today.getMonth() + 1).padStart(2, "0");
    const day = String(today.getDate()).padStart(2, "0");

    dateInput.min = `${year}-${month}-${day}`;


    function checkForm() {

      const selectedFood = document.querySelector(
        'input[name="food"]:checked'
      );

      const hasDate = dateInput.value !== "";
      const hasTime = timeInput.value !== "";

      const complete =
        selectedFood &&
        hasDate &&
        hasTime;

      confirmButton.disabled = !complete;

      if (complete) {
        hint.textContent = "Perfetto. ❤️ Ora puoi confermare.";
      } else {
        hint.textContent =
          "Scegli prima cosa mangiamo e quando. 😏";
      }
    }


    foodInputs.forEach(input => {
      input.addEventListener("change", checkForm);
    });

    dateInput.addEventListener("change", checkForm);
    timeInput.addEventListener("change", checkForm);


    confirmButton.addEventListener("click", () => {

      const selectedFood = document.querySelector(
        'input[name="food"]:checked'
      );

      if (!selectedFood || !dateInput.value || !timeInput.value) {
        return;
      }


      // Formatta la data in italiano
      const date = new Date(
        dateInput.value + "T00:00:00"
      );

      const formattedDate = date.toLocaleDateString(
        "it-IT",
        {
          weekday: "long",
          day: "numeric",
          month: "long",
          year: "numeric"
        }
      );


      finalFood.textContent = selectedFood.value;

      finalDate.textContent =
        formattedDate.charAt(0).toUpperCase() +
        formattedDate.slice(1);

      finalTime.textContent = timeInput.value;


      // Mostra schermata finale
      mainPage.classList.add("hide");
      successPage.classList.add("show");


      // Esplosione di cuori ❤️
      createHearts();

    });


    function createHearts() {

      const container =
        document.getElementById("hearts");

      const emojis = [
        "❤️",
        "💕",
        "💖",
        "💗",
        "💓",
        "🥰"
      ];

      for (let i = 0; i < 35; i++) {

        const heart =
          document.createElement("div");

        heart.className =
          "floating-heart";

        heart.textContent =
          emojis[Math.floor(
            Math.random() * emojis.length
          )];

        heart.style.left =
          Math.random() * 100 + "%";

        heart.style.fontSize =
          (15 + Math.random() * 25) + "px";

        heart.style.animationDuration =
          (3 + Math.random() * 4) + "s";

        heart.style.animationDelay =
          Math.random() * 1.5 + "s";

        container.appendChild(heart);


        setTimeout(() => {
          heart.remove();
        }, 8000);
      }
    }

  </script>
